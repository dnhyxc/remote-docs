# 知识库本地文件夹树

## 1. 背景与目标

知识库「本地文件夹」模式原先将磁盘上的 `.md` 文件以**扁平列表**呈现，当用户选择的目录下存在多级子文件夹时，所有文件平铺展示，缺乏层级感，不便于定位和管理。

本次改动将本地知识库浏览模式升级为**可展开的目录树**，仅展示含有 `.md` 文件的目录路径，实现类文件管理器的层级浏览体验。

## 2. 改动范围

| 文件 | 操作 | 说明 |
|------|------|------|
| `apps/frontend/src/views/knowledge/knowledge-local-tree.ts` | **新增** | 本地 Markdown 目录树数据结构与算法 |
| `apps/frontend/src/views/knowledge/KnowledgeList.tsx` | **修改** | 引入目录树渲染、`KnowledgeFolderRow` 组件、展开/收起状态管理 |

## 3. 实现思路

### 3.1 核心思路

1. **数据树构建**：将后端返回的扁平 `.md` 文件列表，按路径分段重构为嵌套的目录树结构。目录节点仅包含实际含 `.md` 的路径，空目录不展示。
2. **展开状态管理**：使用 `Set<string>` 存储当前展开的目录路径，通过 `useMemo` 派生可见行列表。
3. **渲染策略**：目录行（`KnowledgeFolderRow`）与文件行（`KnowledgeListRow`）分别渲染，通过 `depth` 属性控制缩进层级。
4. **路径归一化**：统一使用正斜杠 `/` 作为路径分隔符，兼容 Windows 反斜杠路径。

### 3.2 为什么不直接渲染文件系统树

- **性能**：无需递归读取磁盘目录结构，仅基于已加载的文件列表构建树，O(n) 时间复杂度。
- **简洁性**：空目录自动过滤，用户只看到有意义的路径层级。
- **兼容性**：与 `invokeListKnowledgeMarkdownFiles` 返回的扁平列表无缝对接。

```typescript
/** 本地知识库：把扁平 .md 列表收成可展开目录树（仅含含有 .md 的路径） */

// 目录节点类型：表示一个文件夹
export type LocalMdTreeDir = {
	type: 'dir';       // 节点类型标记为目录
	name: string;      // 目录名称（不含路径前缀）
	path: string;      // 目录完整路径（已归一化）
	children: LocalMdTreeNode[];  // 子节点列表
};

// 文件节点类型：表示一个 .md 文件
export type LocalMdTreeFile = {
	type: 'file';      // 节点类型标记为文件
	name: string;      // 文件名（不含路径前缀）
	path: string;      // 文件完整路径（已归一化）
	title: string;     // Markdown 标题（用于显示）
	updatedAt: string; // 更新时间（ISO 格式字符串）
};

// 树节点联合类型：目录或文件
export type LocalMdTreeNode = LocalMdTreeDir | LocalMdTreeFile;

// 输入条目类型：构建树所需的最小文件信息
export type LocalMdTreeEntry = {
	path: string;      // 文件完整路径
	title: string;     // Markdown 标题
	updatedAt: string; // 更新时间
};

// 路径归一化：将反斜杠替换为正斜杠，移除末尾斜杠
export function normalizeFsPath(p: string): string {
	return p.replace(/\\/g, '/').replace(/\/+$/, '');
}

// 从路径中提取基础名称（最后一段）
export function basenameFs(p: string): string {
	// 先归一化路径（统一分隔符、去尾斜杠）
	const n = normalizeFsPath(p);
	// 找最后一个斜杠的位置
	const i = n.lastIndexOf('/');
	// 无斜杠则返回全路径，有则返回最后一段
	return i < 0 ? n : n.slice(i + 1) || n;
}
```

#### 4.1.2 树构建算法

```typescript
/** 由根目录 + 扁平 .md 条目构建树；子节点：目录优先，同级按名称排序 */
export function buildLocalMdTree(
	rootDir: string,      // 用户选择的根目录路径
	entries: LocalMdTreeEntry[],  // 扁平 .md 文件列表
): LocalMdTreeDir {
	// 归一化根目录路径，确保格式统一
	const rootPath = normalizeFsPath(rootDir);
	// 创建根目录节点
	const root: LocalMdTreeDir = {
		type: 'dir',                              // 标记为目录
		name: basenameFs(rootPath) || rootPath,    // 取根目录名称
		path: rootPath,                           // 设置根路径
		children: [],                             // 初始化空子节点列表
	};
	// 构建路径前缀，用于判断文件是否在根目录下
	const prefix = `${rootPath}/`;

	// 遍历每个文件条目，插入到树中
	for (const e of entries) {
		// 归一化文件路径
		const filePath = normalizeFsPath(e.path);
		// 跳过不在根目录下的文件
		if (!filePath.startsWith(prefix)) continue;
		// 计算相对路径的分段（按斜杠切分，过滤空段）
		const parts = filePath.slice(prefix.length).split('/').filter(Boolean);
		// 根目录下不应有空路径
		if (parts.length === 0) continue;

		// 从根节点开始，逐层创建/定位中间目录
		let parent = root;
		// parts 的最后一段是文件名，前面的都是目录
		for (let i = 0; i < parts.length - 1; i++) {
			const name = parts[i]!;                    // 当前目录名
			const childPath = `${parent.path}/${name}`;  // 拼接子目录完整路径
			// 在兄弟节点中查找是否已存在该目录
			let child = parent.children.find(
				(c): c is LocalMdTreeDir => c.type === 'dir' && c.path === childPath,
			);
			// 如果目录不存在，创建新目录节点并加入父节点
			if (!child) {
				child = { type: 'dir', name, path: childPath, children: [] };
				parent.children.push(child);
			}
			// 向下一层遍历
			parent = child;
		}

		// parts 的最后一段是文件名，创建文件节点并加入父目录
		const fileName = parts[parts.length - 1]!;
		parent.children.push({
			type: 'file',           // 标记为文件
			name: fileName,         // 文件名
			path: filePath,         // 文件完整路径
			title: e.title,         // Markdown 标题
			updatedAt: e.updatedAt, // 更新时间
		});
	}

	// 对树进行排序（目录优先，同级按名称排序）
	sortDir(root);
	return root;
}
```

#### 4.1.3 树排序与拍平算法

```typescript
// 递归排序目录节点：目录优先于文件，同级按名称（支持中文）排序
function sortDir(dir: LocalMdTreeDir): void {
	// 对当前目录的子节点排序
	dir.children.sort((a, b) => {
		// 类型不同：目录排在文件前面
		if (a.type !== b.type) return a.type === 'dir' ? -1 : 1;
		// 类型相同：按名称做本地化比较（支持中文排序）
		return a.name.localeCompare(b.name, 'zh');
	});
	// 递归排序所有子目录
	for (const c of dir.children) {
		if (c.type === 'dir') sortDir(c);
	}
}

/** 按展开集合拍平成可见行（根节点始终在内） */
export function flattenVisibleLocalMdTree(
	root: LocalMdTreeDir,          // 树的根节点
	expanded: ReadonlySet<string>,  // 当前展开的目录路径集合
): Array<{ node: LocalMdTreeNode; depth: number }> {
	// 输出数组：每个元素包含节点和缩进深度
	const out: Array<{ node: LocalMdTreeNode; depth: number }> = [];
	// 递归遍历函数
	const walk = (node: LocalMdTreeNode, depth: number) => {
		// 将当前节点加入输出
		out.push({ node, depth });
		// 如果是目录且在展开集合中，递归遍历子节点
		if (node.type === 'dir' && expanded.has(node.path)) {
			for (const c of node.children) walk(c, depth + 1);
		}
	};
	// 从根节点开始遍历，深度为 0
	walk(root, 0);
	return out;
}
```

// 新增目录行组件：用于渲染可展开/收起的目录节点
const KnowledgeFolderRow = (props: KnowledgeFolderRowProps) => {
	// 解构 props
	const { node, depth, expanded, onToggle } = props;
	// 键盘事件处理：Enter 或空格键触发展开/收起
	const onKeyDown = (e: React.KeyboardEvent) => {
		if (e.key === 'Enter' || e.key === ' ') {
			e.preventDefault();           // 阻止默认滚动行为
			onToggle(node.path);        // 切换展开状态
		}
	};

	return (
		// 目录行容器：可点击、可聚焦、支持 ARIA 状态
		<div
			role="button"                                  // 语义角色
			tabIndex={0}                                   // 可通过 Tab 聚焦
			aria-expanded={expanded}                       // 无障碍展开状态
			onClick={() => onToggle(node.path)}            // 点击切换
			onKeyDown={onKeyDown}                          // 键盘切换
			// 缩进样式：8px 基础 + 每层 12px
			style={depth > 0 ? { paddingLeft: 8 + depth * 12 } : undefined}
			className="group relative w-full cursor-pointer overflow-hidden flex items-center gap-1.5 p-2 rounded-md transition-colors hover:bg-theme/10"
		>
			// 展开/收起箭头：根据 expanded 状态旋转 90 度
			<ChevronRight
				size={14}
				aria-hidden
				className={cn(
					'shrink-0 text-textcolor/50 transition-transform',
					expanded && 'rotate-90',  // 展开时箭头旋转
				)}
			/>
			// 文件夹图标：展开时显示 FolderOpen，收起时显示 Folder
			{expanded ? (
				<FolderOpen
					size={15}
					className="shrink-0 text-teal-500/90"
					aria-hidden
				/>
			) : (
				<Folder
					size={15}
					className="shrink-0 text-teal-500/90"
					aria-hidden
				/>
			)}
			// 目录名称：截断溢出

	? visibleLocalRows.map(({ node, depth }) => {
		// 目录节点：渲染 KnowledgeFolderRow
		if (node.type === 'dir') {
			return (
				<KnowledgeFolderRow
					key={`dir:${node.path}`}    // 目录节点 key 前缀区分
					node={node}                  // 目录节点数据
					depth={depth}                // 缩进深度
					expanded={expandedDirs.has(node.path)}  // 当前展开状态
					onToggle={toggleLocalDir}    // 展开/收起回调
				/>
			);

		const item = localFileToListItem(node);
		return (
			<KnowledgeListRow
				key={node.path}               // 使用文件路径作为 key
				item={item}                   // 转换后的列表项

				}
				onActivate={handleRowClick}
				onTrashClick={onTrashClick}
				showOpenInExternalEditor={isTauriRuntime()}
				onOpenInExternalEditorClick={
					onOpenInExternalEditorClick
				}
			/>
		);
	})

		/>

## 5. 兼容性与影响

### 5.1 行为变化

| 方面 | 改动前 | 改动后 |
|------|--------|--------|
| 本地文件展示 | 扁平列表 | 可展开目录树 |
| 子目录支持 | 不区分层级 | 多层级缩进展示 |
| 目录展开 | 无此概念 | 点击目录行展开/收起 |
| 空目录展示 | N/A | 自动过滤（仅展示含 .md 的路径） |
| 键盘导航 | 文件行支持 | 目录行同样支持 Enter/Space |

### 5.2 兼容性

- **云端模式不受影响**：仅当 `useLocalFolder = true` 时启用目录树渲染，云端列表保持原有逻辑。
- **API 不变**：`invokeListKnowledgeMarkdownFiles` 返回的扁平列表格式未变，前端内部自行构建树。
- **删除/打开逻辑不受影响**：文件行的操作按钮（删除、在编辑器中打开）在树模式下正常工作。

### 5.3 风险与回归

| 测试路径 | 预期结果 |
|----------|----------|
| 选择单级目录（无嵌套） | 展示根目录 + 所有 .md 文件（缩进 0） |
| 选择多级目录（有嵌套） | 展示根目录 + 子目录 + 文件，可逐层展开 |
| 展开/收起目录 | 点击箭头或目录行，children 显示/隐藏 |
| 选择新文件夹 | 重置展开状态，仅展开新根目录 |
| 文件操作（删除、打开） | 与扁平模式行为一致 |
| 空文件夹 | 显示空状态提示 |
| 非 Tauri 环境 | 本地文件夹模式不可用 |
