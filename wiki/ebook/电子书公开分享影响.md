## 1. 背景与目标

### 1.1 用户故事

| 角色 | 行为 | 期望 |
|------|------|------|
| 书主 A | 书架/阅读页将 EPUB 切 **公开** | 全站登录用户可在书架看到 |
| 读者 B | 书架「全部」或「公开」Tab 点进 A 的书 | 进入 `/ebook/read/:readingBookId`，无需链接 |
| 读者 B | 阅读时写想法、划线 | 仅 B 可见/可改；可见 A 的 **公开** 想法 |
| 书主 A | 取消公开 | B 书架不再展示；B 已有读书记录保留 |

### 1.2 数据模型

| 实体 | 字段 | 含义 |
|------|------|------|
| 源书 | `is_public`, `source_book_id IS NULL` | 书主上架 EPUB |
| 读书记录 | `source_book_id → 源书` | 读者派生书，存 B 的进度/批注 |
| 想法 | `is_public` | 读者想法默认公开（书主可见）；过滤见 `filterThoughtsVisibleToUser` |

### 1.3 改动范围

| 层级 | 路径 |
|------|------|
| 迁移 | `apps/backend/src/migrations/*ebook_share*`、`*ebook_auth*` |

## 2. 实现思路

| # | 要点 | 理由 |
|---|------|------|
| 1 | **源书 + 读书记录** | 进度/划线/想法按 `userId`+`bookId` 隔离，共享 COS `filePath` |
| 2 | **`openPublicBook` 幂等** | 已有读书记录则返回 `readingBookId`，不重复建 |
| 3 | **书架 `scope=public`** | 查他人 `is_public` 源书，附带 `owner` 与 `readingBookId` |
| 4 | **`listThoughts` 三分支** | 读书记录：源书主想法 + 本人；源书公开：主想法 + 各读者公开想法 |
| 5 | **仅 store EPUB 可公开** | 本地 path 书须先上云；PDF 不支持 |
| 6 | **前端 `openEbookPublicBook`** | 点公开书卡片时先 open 再 navigate 到读书记录 id |

```typescript
// 书主切换源书公开/私有
async setBookVisibility(
	userId: number,
	bookId: string,
	isPublic: boolean,
): Promise<EbookBookDto> {
	// 仅书主自己的书
	const book = await this.bookRepo.findOne({
		where: { id: bookId, userId },
	});
	if (!book) {
		throw new NotFoundException('书籍不存在');
	}
	// 读书记录不能改公开态
	if (book.sourceBookId) {
		throw new ForbiddenException('读书记录不能设置公开状态');
	}
	// 一期仅 EPUB
	if (book.fmt !== 'epub') {
		throw new BadRequestException('仅支持公开 EPUB 书籍');
	}
	// 公开前须已上云（store + filePath）
	if (isPublic && (!book.filePath || book.srcKind !== 'store')) {
		throw new BadRequestException('请先上传至云端后再公开');
	}
	book.isPublic = isPublic;
	// 首次公开写入 public_at
	if (isPublic && !book.publicAt) {
		book.publicAt = new Date();
	}
	await this.bookRepo.save(book);
	return this.toBookDto(book);
}
```

**变更摘要**：校验源书/格式/云端后写 `isPublic` 与 `publicAt`。

```typescript
// 读者从公开源书创建或复用读书记录
async openPublicBook(
	userId: number,
	sourceBookId: string,
): Promise<{ readingBookId: string }> {
	// 源书须存在、公开、且为源书行
	const source = await this.bookRepo.findOne({
		where: { id: sourceBookId, sourceBookId: IsNull(), isPublic: true },
	});
	if (!source) {
		throw new NotFoundException('书籍不存在或已下架');
	}
	// 不能 open 自己的书（应直接读源书 id）
	if (source.userId === userId) {
		throw new BadRequestException('不能通过公开入口打开自己的书');
	}
	if (source.fmt !== 'epub' || !source.filePath) {
		throw new BadRequestException('仅支持公开 EPUB 书籍');
	}

	// 幂等：已有读书记录则直接返回
	const existing = await this.bookRepo.findOne({
		where: { userId, sourceBookId },
	});
	if (existing) {
		return { readingBookId: existing.id };
	}

	// 克隆元数据与 filePath，不复制 COS 文件
	const reading = this.bookRepo.create({
		userId,
		sourceBookId,
		fmt: source.fmt,
		title: source.title,
		author: source.author,
		srcKind: 'store',
		filePath: source.filePath,
		size: source.size,
		coverPath: source.coverPath,
		isPublic: false,
	});
	await this.bookRepo.save(reading);
	return { readingBookId: reading.id };
}
```

**变更摘要**：公开源书 → 读者派生书；共享 `filePath`；返回读书记录 id 供阅读页使用。

```typescript
async listThoughts(userId: number, bookId: string): Promise<EbookThoughtDto[]> {
	const book = await this.assertBookOwned(userId, bookId);
	let rows: EbookThought[];

	// 读者读书记录：合并源书主想法 + 本人想法
	if (book.sourceBookId) {
		const source = await this.bookRepo.findOne({
			where: { id: book.sourceBookId },
		});
		if (!source) {
			throw new NotFoundException('源书不存在');
		}
		// 源书已下架时仍允许已有读书记录继续读
		if (!source.isPublic) {
			const hasRecord = await this.bookRepo.findOne({
				where: { userId, sourceBookId: book.sourceBookId },
			});
			if (!hasRecord) {
				throw new NotFoundException('书籍不存在');
			}
		}
		const ownerRows = await this.thoughtRepo.find({
			where: { bookId: source.id, userId: source.userId },
		});
		const ownRows = await this.thoughtRepo.find({
			where: { bookId, userId },
		});
		rows = [...ownerRows, ...ownRows].sort(
			(a, b) => b.createdAt.getTime() - a.createdAt.getTime(),
		);
	} else if (book.isPublic) {
		// 书主读自己的公开源书：主想法 + 各读者公开想法
		const ownerRows = await this.thoughtRepo.find({
			where: { bookId, userId },
		});
		const readingIds = (
			await this.bookRepo.find({
				where: { sourceBookId: book.id },
				select: ['id'],
			})
		).map((r) => r.id);
		const readerRows =
			readingIds.length === 0
				? []
				: await this.thoughtRepo.find({
					where: { bookId: In(readingIds) },
				});
		rows = [...ownerRows, ...readerRows].sort(
			(a, b) => b.createdAt.getTime() - a.createdAt.getTime(),
		);
	} else {
		// 私有书：仅本人
		rows = await this.thoughtRepo.find({
			where: { userId, bookId },
			order: { createdAt: 'DESC' },
		});
	}

	// 过滤非公开/不可见想法
	rows = this.filterThoughtsVisibleToUser(rows, userId);
	// ... buildUserInfoMap + toThoughtDtoWithUsername
}
```

**变更摘要**：三分支合并书主/读者想法；前端 `currentUserId` 驱动 mark 色与叠层（见 overlay 专题）。

```typescript
/** 书主将 EPUB 设为公开 / 私有 */
export function EbookBookVisibilitySwitch({
	book,
	canToggle = true,
	className,
	compact = false,
}: Props) {
	const { t } = useI18n();
	const [confirmOpen, setConfirmOpen] = useState(false);
	const [pendingPublic, setPendingPublic] = useState<boolean | null>(null);
	const [busy, setBusy] = useState(false);

	const isPublic = Boolean(book.isPublic);
	// 读书记录、非 EPUB、无权限或请求中禁用
	const disabled =
		!canToggle || book.fmt !== 'epub' || book.sourceBookId != null || busy;

	const requestToggle = useCallback(
		(next: boolean) => {
			if (disabled) return;
			setPendingPublic(next);
			setConfirmOpen(true);
		},
		[disabled],
	);

	const onConfirm = useCallback(async () => {
		if (pendingPublic == null) return;
		setBusy(true);
		try {
			await ebookStore.setBookPublic(book.id, pendingPublic);
			setConfirmOpen(false);
			setPendingPublic(null);
		} catch (e) {
			Toast({ type: 'error', title: t('ebook.public.visibilityFailed'), message: getRequestErrorMessage(e) });
		} finally {
			setBusy(false);
		}
	}, [book.id, pendingPublic, t]);

	// compact：书架卡片仅 Globe/Lock 图标；非 compact：阅读页带文案按钮
	// ... Confirm + Tooltip + Button 渲染
}
```
